cmake_minimum_required(VERSION 3.22)

# ==============================================================================
# 推荐的项目物理结构
# ==============================================================================
#
# 框架搭建与测试/
# │
# ├── CMakeLists.txt
# │
# ├── include/                    # C 语言头文件
# │   ├── demo.h
# │   └── ...
# │
# ├── src/                        # C 语言源文件
# │   ├── main.c                  # 程序入口
# │   ├── demo.c
# │   └── ...
# │
# └── build -> ...                # 构建目录软链接
#
#
# 当前 ubuntu-asus 的构建策略：
#
#   源码：
#       保留在 Ubuntu 本地文件系统
#
#   build：
#       项目中的 build 是符号链接，
#       真实目录位于：
#
#           /mnt/buildssd/builds/<project>
#
#       /mnt/buildssd 通过 SMB/CIFS 映射到 Windows SSD。
#
# 这样可以让源码继续保留在 Linux 本地，
# 同时把构建过程中较频繁的磁盘读写放到 SSD。
#
#
# 项目首次准备时，可在项目根目录执行：
#
#   fastbuild-init
#
# 该脚本会创建对应的真实构建目录，
# 并建立项目内的 build 符号链接。
#
#
# 标准配置方式：
#
#   cmake -S . -B build -G Ninja
#
# 标准构建方式：
#
#   cmake --build build
#
#
# 注意：
#
#   不要使用：
#
#       cd build
#       cmake ..
#
#   因为 build 是符号链接。
#
#   进入 build 后，".." 可能按照真实文件系统路径解析，
#   此时它并不一定指向项目源码目录。
#
#   因此统一使用 CMake 的 -S / -B 形式，
#   显式指定源码目录和构建目录。
#
# ==============================================================================


# ==============================================================================
# 1. 工具链控制
# ==============================================================================
#
# 优先读取环境变量 CC：
#
#   export CC=clang-21
#
# 如果环境变量没有指定，则默认使用 clang-21。
#
# CMAKE_C_COMPILER 必须在 project(... LANGUAGES C) 之前确定，
# 因为 project() 启用 C 语言时会立即进行编译器探测。
#
# ==============================================================================

if(DEFINED ENV{CC})
    set(CMAKE_C_COMPILER "$ENV{CC}")
else()
    set(CMAKE_C_COMPILER clang-21)
endif()


# ==============================================================================
# 2. 项目定义
# ==============================================================================

project(
    C_Basics_Framework
    VERSION 1.0
    LANGUAGES C
)


# ==============================================================================
# 3. 编译器检查
# ==============================================================================
#
# 本项目统一使用 Clang。
#
# 当前学习机采用 Clang 21 工具链，因此：
#
#   - 如果不是 Clang，终止配置；
#   - 如果 Clang 版本低于 21，终止配置。
#
# 这样可以避免不同机器或不同工具链造成不必要的构建差异。
#
# ==============================================================================

if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(
        FATAL_ERROR
        "当前编译器不是 Clang。\n"
        "请使用:export CC=clang-21"
    )
endif()

if(CMAKE_C_COMPILER_VERSION VERSION_LESS 21)
    message(
        FATAL_ERROR
        "Clang 版本过低。\n"
        "当前版本：${CMAKE_C_COMPILER_VERSION}\n"
        "要求版本:21 或更高"
    )
endif()


# ==============================================================================
# 4. C 语言标准
# ==============================================================================
#
# 本项目使用 C17。
#
# CMAKE_C_STANDARD_REQUIRED ON
# ------------------------------------------------------------------------------
# 要求编译器真正支持指定的语言标准。
#
# 如果无法满足 C17，
# CMake 不允许静默降级到更早版本。
#
#
# CMAKE_C_EXTENSIONS OFF
# ------------------------------------------------------------------------------
# 禁止默认启用 GNU 等编译器提供的非标准语言扩展。
#
# 例如 Clang/GCC 原本可能采用：
#
#   -std=gnu17
#
# 关闭扩展后则使用标准 C：
#
#   -std=c17
#
# 对基础复习来说，可以更清楚地区分：
#
#   ISO C 标准本身
#
# 与
#
#   某个编译器额外提供的扩展。
#
# ==============================================================================

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)


# ==============================================================================
# 5. 导出 compile_commands.json
# ==============================================================================
#
# 在构建目录生成：
#
#   compile_commands.json
#
# 其中记录各源文件真正使用的编译命令，包括：
#
#   - 编译器
#   - include 搜索路径
#   - 编译选项
#   - 宏定义
#   - C 语言标准
#
# clangd、VS Code 及其他静态分析工具可以利用它获得
# 更准确的代码补全、诊断和跳转信息。
#
# ==============================================================================

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)


# ==============================================================================
# 6. 编译警告
# ==============================================================================
#
# -Wall
# ------------------------------------------------------------------------------
# 开启 Clang/GCC 中一组常用且价值较高的警告。
#
# 注意：
#
#   -Wall 并不是“开启所有警告”。
#
# 它主要用于发现常见问题，例如：
#
#   - 未使用的变量；
#   - 部分可疑表达式；
#   - 某些控制流问题；
#   - 一些常见的类型使用问题。
#
#
# -Wextra
# ------------------------------------------------------------------------------
# 在 -Wall 的基础上进一步开启额外警告。
#
# 常见用途包括发现：
#
#   - 未使用的函数参数；
#   - 更多值得怀疑的表达式；
#   - 一些 -Wall 默认不会报告的问题。
#
#
# -Wpedantic
# ------------------------------------------------------------------------------
# 对 ISO C 标准之外的语言扩展进行更严格的检查。
#
# 某些代码虽然 Clang 能够正常接受，
# 但并不意味着它属于标准 C。
#
# 这一选项可以帮助区分：
#
#   “C 标准允许这样写”
#
# 和
#
#   “Clang 恰好允许这样写”。
#
#
# 当前不启用 -Werror。
#
# -Werror 会把所有警告提升为编译错误。
#
# 对 CI 或要求严格的正式项目很有价值，
# 但基础学习和实验阶段可能因此被非关键警告打断，
# 所以暂时只报告警告，不强制失败。
#
# ==============================================================================

add_compile_options(
    -Wall
    -Wextra
    -Wpedantic
)


# ==============================================================================
# 7. 项目文件自动发现
# ==============================================================================
#
# 本项目采用：
#
#   “构建规则保持稳定，具体文件自动发现”
#
# 的管理方式。
#
# CMakeLists.txt 负责描述长期有效的工程规则，
# 日常增加、删除或重命名普通源文件时，
# 原则上不需要修改本文件。
#
#
# src/
# ------------------------------------------------------------------------------
#
# 自动递归发现：
#
#   *.c
#
#
# include/
# ------------------------------------------------------------------------------
#
# 自动递归发现：
#
#   *.h
#
#
# 当前是纯 C 项目，因此有意不自动收集：
#
#   *.cc
#   *.cpp
#   *.cxx
#   *.hpp
#
# 避免 C 语言复习工程无意间混入 C++ 文件。
#
#
# CONFIGURE_DEPENDS
# ------------------------------------------------------------------------------
#
# 普通 file(GLOB ...) 通常只在配置阶段扫描文件。
#
# CONFIGURE_DEPENDS 会让构建系统检查匹配文件集合是否发生变化。
#
# 因此新增、删除或重命名：
#
#   src/example.c
#   include/example.h
#
# 后，在后续构建过程中 CMake 可以重新配置并更新文件列表。
#
# 这样新增普通编译单元不需要反复维护 CMakeLists.txt。
#
# ==============================================================================

file(
    GLOB_RECURSE
    PROJECT_SOURCES
    CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c"
)

file(
    GLOB_RECURSE
    PROJECT_HEADERS
    CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h"
)


# ==============================================================================
# 8. 基本文件完整性检查
# ==============================================================================
#
# 当前工程最终生成一个 C 可执行程序，
# 因此至少必须存在一个 .c 源文件。
#
# 如果 src/ 中没有找到任何 .c 文件，
# 说明项目尚不具备基本的构建条件，
# 在配置阶段直接终止。
#
# 头文件并不是 C 程序的必需组成部分，
# 所以 PROJECT_HEADERS 允许为空。
#
# ==============================================================================

if(NOT PROJECT_SOURCES)
    message(
        FATAL_ERROR
        "src/ 目录中没有找到任何 .c 源文件。"
    )
endif()


# ==============================================================================
# 9. 创建可执行目标
# ==============================================================================
#
# PROJECT_SOURCES
# ------------------------------------------------------------------------------
#
# 保存真正参与编译的 .c 源文件。
#
#
# PROJECT_HEADERS
# ------------------------------------------------------------------------------
#
# .h 文件不会作为独立翻译单元进行编译。
#
# 将其列入 target，主要用于：
#
#   - 明确这些头文件属于当前目标；
#   - 改善部分 IDE / 编辑器中的工程文件视图；
#   - 让 target 的文件组成更加完整。
#
# 真正决定 #include "xxx.h" 能否找到头文件的，
# 是后面的 target_include_directories()。
#
# ==============================================================================

add_executable(
    main

    ${PROJECT_SOURCES}
    ${PROJECT_HEADERS}
)


# ==============================================================================
# 10. 头文件搜索路径
# ==============================================================================
#
# 将项目的：
#
#   include/
#
# 加入 main 的头文件搜索路径。
#
# 因此源码中可以：
#
#   #include "demo.h"
#
# 而不需要写：
#
#   #include "../include/demo.h"
#
#
# PRIVATE
# ------------------------------------------------------------------------------
#
# 表示该 include 路径仅属于 main 当前目标，
# 不会自动传播给其他依赖目标。
#
# 当前项目只有一个可执行目标，
# 因此使用 PRIVATE。
#
# ==============================================================================

target_include_directories(
    main
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/include"
)


# ==============================================================================
# 11. 配置信息输出
# ==============================================================================
#
# 在 CMake 配置阶段输出一份简洁的环境摘要，
# 用于快速确认当前实际使用的：
#
#   - 编译器；
#   - 编译器版本；
#   - C 标准；
#   - 源码目录；
#   - 构建目录；
#   - 自动发现的项目文件。
#
# 对远程 SSH 开发以及跨机器构建环境而言，
# 这些信息可以帮助快速发现环境配置异常。
#
# ==============================================================================

message(STATUS "")
message(STATUS "==================================================")
message(STATUS " 项目              : ${PROJECT_NAME}")
message(STATUS " 项目版本          : ${PROJECT_VERSION}")
message(STATUS " C 编译器          : ${CMAKE_C_COMPILER}")
message(STATUS " 编译器 ID         : ${CMAKE_C_COMPILER_ID}")
message(STATUS " 编译器版本        : ${CMAKE_C_COMPILER_VERSION}")
message(STATUS " C 标准            : C${CMAKE_C_STANDARD}")
message(STATUS " C 扩展            : ${CMAKE_C_EXTENSIONS}")
message(STATUS " 源码目录          : ${CMAKE_SOURCE_DIR}")
message(STATUS " 构建目录          : ${CMAKE_BINARY_DIR}")
message(STATUS " 生成器            : ${CMAKE_GENERATOR}")
message(STATUS " 可执行目标        : main")
message(STATUS "")
message(STATUS " C 源文件          : ${PROJECT_SOURCES}")
message(STATUS " C 头文件          : ${PROJECT_HEADERS}")
message(STATUS "==================================================")
message(STATUS "")
