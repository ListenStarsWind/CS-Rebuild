# ==============================================================================
# CTest 测试子项目
# ==============================================================================
#
# 测试程序拥有自己的 main()。
#
# 自动复用项目 src/ 中除 main.c 外的全部实现文件，
# 因此不需要手工维护：
#
#   bubble_sort.c
#   int_swap.c
#   ...
#
# 后续 src/ 中新增实现文件时，也会自动进入测试目标。
#
# ==============================================================================



# ==============================================================================
# 1. 准备测试所需的项目源码
# ==============================================================================


set(
    PROJECT_TEST_SOURCES
    ${PROJECT_SOURCES}
)


list(
    REMOVE_ITEM
    PROJECT_TEST_SOURCES
    "${PROJECT_SOURCE_DIR}/src/main.c"
)



# ==============================================================================
# 2. 创建 C 测试目标
# ==============================================================================


function(add_c_test test_name)


    add_executable(

        ${test_name}

        ${ARGN}

        ${PROJECT_TEST_SOURCES}

        ${PROJECT_HEADERS}

    )



    # ------------------------------
    # 项目头文件路径
    # ------------------------------

    target_include_directories(

        ${test_name}

        PRIVATE

        "${PROJECT_SOURCE_DIR}/include"

    )



    # ------------------------------
    # 编译参数
    # ------------------------------

    target_compile_options(

        ${test_name}

        PRIVATE

        ${PROJECT_COMPILE_OPTIONS}

    )



    # ------------------------------
    # 链接库
    # ------------------------------

    target_link_libraries(

        ${test_name}

        PRIVATE

        ${PROJECT_LIBRARIES}

    )



    # ------------------------------
    # 注册到 CTest
    # ------------------------------

    add_test(

        NAME ${test_name}

        COMMAND ${test_name}

    )


endfunction()



# ==============================================================================
# 3. 冒泡排序测试
# ==============================================================================


add_c_test(
    test_bubble_sort
    test_bubble_sort.c
)
